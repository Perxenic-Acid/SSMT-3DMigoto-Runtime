cmake_minimum_required(VERSION 3.20)
project(context_interop_test LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "此测试需要 Windows D3D11。")
endif()
add_executable(context-interop-test main.cpp)
target_compile_features(context-interop-test PRIVATE cxx_std_17)
target_link_libraries(context-interop-test PRIVATE d3d11 d3dcompiler)
if(MSVC)
    target_compile_options(context-interop-test PRIVATE /utf-8)
endif()

include(CTest)
add_executable(context-upgrade-test upgrade.cpp)
target_compile_features(context-upgrade-test PRIVATE cxx_std_17)
target_compile_options(context-upgrade-test PRIVATE /utf-8)
target_link_libraries(context-upgrade-test PRIVATE d3d11)
add_executable(texture-interop-test textures.cpp)
target_compile_features(texture-interop-test PRIVATE cxx_std_17)
target_compile_options(texture-interop-test PRIVATE /utf-8)
target_link_libraries(texture-interop-test PRIVATE d3d11)
add_executable(params-interop-test params.cpp)
target_compile_features(params-interop-test PRIVATE cxx_std_17)
target_compile_options(params-interop-test PRIVATE /utf-8)
target_link_libraries(params-interop-test PRIVATE d3d11 d3dcompiler)
set(SSMT_RUNTIME_DLL "${CMAKE_CURRENT_SOURCE_DIR}/../../x64/Release/d3d11.dll" CACHE FILEPATH "待测的 x64 SSMT Runtime")
if(BUILD_TESTING)
    add_custom_target(stage-context-runtime ALL)
    foreach(mode IN ITEMS early delayed)
        set(params_stage "${CMAKE_CURRENT_BINARY_DIR}/params-${mode}")
        file(MAKE_DIRECTORY "${params_stage}")
        add_custom_command(TARGET stage-context-runtime POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SSMT_RUNTIME_DLL}" "${params_stage}/d3d11.dll")
        add_test(NAME "params-${mode}" COMMAND ${CMAKE_COMMAND}
            "-DFIXTURE=$<TARGET_FILE:params-interop-test>" "-DRUNTIME=${params_stage}/d3d11.dll" "-DMODE=${mode}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/RunParams.cmake")
        set_tests_properties("params-${mode}" PROPERTIES TIMEOUT 30)
    endforeach()
    set(upgrade_stage "${CMAKE_CURRENT_BINARY_DIR}/upgrade")
    file(MAKE_DIRECTORY "${upgrade_stage}")
    file(WRITE "${upgrade_stage}/d3dx.ini" "[Loader]\ntarget=context-upgrade-test.exe\n[System]\nload_library_redirect=2\nallow_platform_update=1\n[Hunting]\nhunting=0\n[Logging]\ncalls=1\nunbuffered=1\n[ClearRenderTargetView]\npost clear=this 1 0 0 1\n")
    add_custom_command(TARGET stage-context-runtime POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SSMT_RUNTIME_DLL}" "${upgrade_stage}/d3d11.dll")
    add_test(NAME context-upgrade COMMAND context-upgrade-test "${upgrade_stage}/d3d11.dll")
    set_tests_properties(context-upgrade PROPERTIES TIMEOUT 30)
    set(texture_stage "${CMAKE_CURRENT_BINARY_DIR}/textures")
    file(MAKE_DIRECTORY "${texture_stage}")
    # 固定 RGBA8 内容及旧版描述哈希；同一夹具同时验证旧入口和 Device3 入口。
    file(WRITE "${texture_stage}/d3dx.ini" "[Loader]\ntarget=texture-interop-test.exe\n[System]\nload_library_redirect=2\nallow_platform_update=1\n[Rendering]\ntexture_hash=0\n[Hunting]\nhunting=0\n[Logging]\ncalls=1\nunbuffered=1\n[ClearRenderTargetView]\npost checktextureoverride=ps-t0\n[ResourceGreen]\nfilename=green.dds\n[TextureOverride2D]\nhash=d20f0658\npost ps-t0=ResourceGreen\n[TextureOverride3D]\nhash=00d7eba0\npost ps-t0=ResourceGreen\n[TextureOverrideResize]\nhash=565fcf81\nwidth=9\n")
    add_custom_command(TARGET stage-context-runtime POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SSMT_RUNTIME_DLL}" "${texture_stage}/d3d11.dll")
    add_test(NAME texture-interop COMMAND ${CMAKE_COMMAND}
        "-DFIXTURE=$<TARGET_FILE:texture-interop-test>" "-DRUNTIME=${texture_stage}/d3d11.dll"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/RunTextures.cmake")
    set_tests_properties(texture-interop PROPERTIES TIMEOUT 30)
    foreach(hunting IN ITEMS 0 2)
        foreach(count IN ITEMS 0 128 134)
            set(test_name "hunting-${hunting}-clone-${count}")
            set(stage "${CMAKE_CURRENT_BINARY_DIR}/cases/${test_name}")
            file(MAKE_DIRECTORY "${stage}")
            file(WRITE "${stage}/d3dx.ini" "[Loader]\ntarget=context-interop-test.exe\n[System]\nload_library_redirect=2\n[Device]\nallow_platform_update=1\n[Hunting]\nhunting=${hunting}\nanalyse_frame=VK_F8\n[Logging]\ncalls=1\nunbuffered=1\n[ClearRenderTargetView]\nanalyse_options=mono\n")
            add_custom_command(TARGET stage-context-runtime POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SSMT_RUNTIME_DLL}" "${stage}/d3d11.dll")
            add_test(NAME "${test_name}" COMMAND ${CMAKE_COMMAND}
                "-DFIXTURE=$<TARGET_FILE:context-interop-test>" "-DRUNTIME=${stage}/d3d11.dll" "-DCOUNT=${count}"
                -P "${CMAKE_CURRENT_SOURCE_DIR}/RunCase.cmake")
            set_tests_properties("${test_name}" PROPERTIES TIMEOUT 30)
        endforeach()
    endforeach()
    foreach(api IN ITEMS Present Present1 Deferred)
        set(stage "${CMAKE_CURRENT_BINARY_DIR}/capture-${api}")
        file(MAKE_DIRECTORY "${stage}")
        file(WRITE "${stage}/d3dx.ini" "[Loader]\ntarget=context-interop-test.exe\n[System]\nload_library_redirect=2\nconfig_initialization_delay=-1\ncheck_foreground_window=0\n[Device]\nallow_platform_update=1\n[Hunting]\nhunting=1\nanalyse_frame=no_VK_F24\nanalyse_options=dump_vb buf txt mono deferred_ctx_immediate\n[Logging]\ncalls=1\nunbuffered=1\n")
        add_custom_command(TARGET stage-context-runtime POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SSMT_RUNTIME_DLL}" "${stage}/d3d11.dll")
        add_test(NAME "capture-${api}" COMMAND ${CMAKE_COMMAND}
            "-DFIXTURE=$<TARGET_FILE:context-interop-test>" "-DRUNTIME=${stage}/d3d11.dll" "-DAPI=${api}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/RunCapture.cmake")
        set_tests_properties("capture-${api}" PROPERTIES TIMEOUT 30)
    endforeach()
endif()
