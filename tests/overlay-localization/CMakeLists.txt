cmake_minimum_required(VERSION 3.20)
project(overlay_localization LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "此测试需要 Windows 系统字体与 D3D11 WARP。")
endif()
set(RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(DIRECTXTK_LIBRARY "${RUNTIME_ROOT}/DirectXTK/Bin/Desktop_2017/x64/Release/DirectXTK.lib" CACHE FILEPATH "Runtime 构建的 x64 Release DirectXTK")
add_executable(overlay-localization-test main.cpp
    "${RUNTIME_ROOT}/DirectX11/SSMTOverlayText.cpp"
    "${RUNTIME_ROOT}/DirectX11/SSMTOverlayFont.cpp")
target_compile_features(overlay-localization-test PRIVATE cxx_std_14)
set_property(TARGET overlay-localization-test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_compile_definitions(overlay-localization-test PRIVATE NOMINMAX)
target_compile_options(overlay-localization-test PRIVATE /utf-8)
target_include_directories(overlay-localization-test PRIVATE "${RUNTIME_ROOT}/DirectX11" "${RUNTIME_ROOT}/DirectXTK/Inc")
target_link_libraries(overlay-localization-test PRIVATE "${DIRECTXTK_LIBRARY}" d3d11 gdi32)
include(CTest)
add_test(NAME overlay-localization COMMAND overlay-localization-test)
set_tests_properties(overlay-localization PROPERTIES TIMEOUT 30)
set(SSMT_RUNTIME_DLL "${RUNTIME_ROOT}/x64/Release/d3d11.dll" CACHE FILEPATH "待测的 x64 Runtime DLL")
add_executable(overlay-runtime-test runtime.cpp)
target_compile_features(overlay-runtime-test PRIVATE cxx_std_14)
target_compile_options(overlay-runtime-test PRIVATE /utf-8)
target_link_libraries(overlay-runtime-test PRIVATE d3d11)
add_custom_target(stage-overlay-runtime ALL)
foreach(language IN ITEMS zh_s en)
    set(stage "${CMAKE_CURRENT_BINARY_DIR}/runtime-${language}")
    file(MAKE_DIRECTORY "${stage}")
    file(WRITE "${stage}/d3dx.ini" "[Loader]\ntarget=overlay-runtime-test.exe\n[System]\nload_library_redirect=2\nallow_platform_update=1\ncheck_foreground_window=0\nconfig_initialization_delay=-1\n[Hunting]\nhunting=1\n[Include]\ninclude=overlay-language.ini\n[Logging]\nlog_level=info\n")
    file(WRITE "${stage}/overlay-language.ini" "[Hunting]\noverlay_language=${language}\n")
    add_custom_command(TARGET stage-overlay-runtime POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${SSMT_RUNTIME_DLL}" "${stage}/d3d11.dll")
    add_test(NAME "overlay-runtime-${language}" COMMAND ${CMAKE_COMMAND}
        "-DFIXTURE=$<TARGET_FILE:overlay-runtime-test>" "-DRUNTIME=${stage}/d3d11.dll"
        "-DSTAGE=${stage}" "-DLANGUAGE=${language}" -P "${CMAKE_CURRENT_SOURCE_DIR}/RunRuntime.cmake")
    set_tests_properties("overlay-runtime-${language}" PROPERTIES TIMEOUT 30)
endforeach()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME translation-format-contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_formats.py")
endif()
